\section{Suprafixes}

\begin{frame}[fragile]{Suprafixes}
Suprafix: suprasegmental affix (e.g. tone, stress)\\
Tonal affixes in the case of Cantonese\\
\medskip
\textbf{tonal affix ≠ tone-sandhi}\quad {\footnotesize although both are called 變調 in Chinese}\\
\underline{Tone-sandhi}:\\
\textsuperscript{(largely)} automatic \& predictable process of tones changing, \\
\quad given a particular phonological / syntactic environment\\
\medskip
e.g. Standard Mandarin tone 3 sandhi rule\\
買馬 mǎi mǎ > mái mǎ (= 埋馬 mái mǎ)\\
\smallskip
e.g. Shanghainese\\
三杯 se˥˧ pe˥˧ > se˥ pe˨˩\\
四杯 sz˧˦ pe˥˧ > sz˧ pe˦\\ 

\end{frame}

\begin{frame}[fragile]{Suprafixes}
Tone Sandhi in Cantonese?\\
e.g. Cheung 張 (1969)\\ 
differing sandhi rules of ˥˧ > ˥ in Canton vs Hong Kong\\
\medskip
\emph{However, it is very hard to find people nowadays who contrast ˥˧ and ˥ consistently\\
\quad Hong Kong is nowadays basically all ˥}\\
\smallskip
Most Cantonese speakers have no tone sandhi

\vfill \setstretch{0.65} {\tiny Cheung, Yat-Shing 張日昇. 1969. 香港粵語陰平調及變調問題 A study on the upper
even tone and changed tones in Cantonese as spoken in Hong Kong. \emph{The Journal of The Institute of Chinese Studies of The Chinese
University of Hong Kong} 香港中文大學中國文化研究所學報 2: 81–107. (\url{https://www.cuhk.edu.hk/ics/journal/articles/v02p081.pdf}).} 

\end{frame}

\begin{frame}[fragile]{Suprafixes}
\textbf{\underline{Suprafix}} {\scriptsize (or more-complex affix with a floating tone)}:\\
{\small \emph{(Cantonese suprafixes are tone-substitutive)}}\\
\medskip
e.g. verbal:\\
\medskip
食 sik6 \enquote*{eat} > sik6*2 (= sik6 zo2 食咗  eat \textsc{pfv})\\
\smallskip
nei5 sik6*2 faan6 mei6 aa3?\\
\textsc{2sg} eat:\textsc{pfv} rice not.yet \textsc{q}\\
\enquote*{Have you eaten?}\\
\bigskip
氹 tam3 \enquote*{to coax}, \enquote*{to cheer s.o. up}\\
tam3*2\~{}tam3 (= tam3 jat1 tam3 氹一氹 coax:\textsc{delim})\\
\smallskip
tam3*2\~{}tam3 keoi5\\
coax:\textsc{delim} \textsc{3sg}\\
\enquote*{Coax him/her a bit.}

\end{frame}
