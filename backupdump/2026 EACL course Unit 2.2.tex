\section{Middle Chinese}

\begin{frame}[fragile]{Middle Chinese}
The vast majority of linguistic material in Cantonese can be traced to Middle Chinese\\
\bigskip
\enquote{Early Middle Chinese}:\\
切韻 cit3 wan5 / qièyùn (rime book; 601 隋)\\
\smallskip
\quad {\footnotesize Only partially survived; but there are many augmented editions}\\
\quad {\footnotesize e.g. 廣韻 gwong2 wan5 / guǎngyùn (1008; official rime book of 宋)}\\
\bigskip
author of 切韻: 陸法言 luk6 faat3 jin4 / lù fǎyán \textsuperscript{鮮卑} \\  
\bigskip
{\footnotesize 隋 Suí (581-618) unified China in 589 after \(\approx \)300 years of disunity\\
\medskip
\quad eight scholars (friends of 陸法言's father) decided on the \enquote{best} pronunciation of characters (i.e. making a unified standard) based on a compromise of the old northern and southern standards\\}
\end{frame}

e.g.\\
/taj\textsuperscript{C}/ [taj˧]\\
/daj\textsuperscript{C}/ [daj˧]\\
\medskip
tone-splitting may occur:\\
/taj\textsuperscript{C}/ [taj˧]\\
/daj\textsuperscript{C}/ [daj˨] {\scriptsize (voiced onset lowering pitch)}\\
\medskip
\textit{\scriptsize (In most, but not all, Sinitic languages, voiced initials have become devoiced)}\\
tone-splitting becomes phonemicised:\\
/taj\textsuperscript{C1}/ /taj˧/ (e.g. Canto. 帶 daai3)\\
/taj\textsuperscript{C2}/ /taj˨/ (e.g. Canto. 大 daai6)\\
\medskip
cf. Cantonese tone 3 /˧/ 6 /˨/ both descended from MC tone C\\
cf. Standard Mandarin where MC tone C has not split:\\
\quad 帶 dài, 大 dài (as in 大夫 dài fu)

\section{MC initials}

\begin{frame}[fragile]{MC initials \& Late Middle Chinese}
Early Middle Chinese:《切韻》 cit3 wan5 / qièyùn (601 隋)\\ (artificial: compromise between N and S Standard Chinese)\\
\smallskip
organised by tones and finals\\ 
no organised information on the initials\\
\bigskip
Late Middle Chinese: rime tables 韻圖 wan5 tou4 / yùntú\\
\smallskip
《韻鏡》 wan5 geng3 / yùnjìng $\approx$10\textsuperscript{th} c 五代; 1161 南宋 ed.)\\
《七音略》 cat1 jam1 loek6 / qīyīnlüè (before 1161 南宋)\\
\smallskip
information on tones and finals: basically the same as《切韻》\\
information on initials: current (10\textsuperscript{th} century)\\
\medskip
{\footnotesize (There are rime tables like《四聲等子》(1039 北宋) and《切韻指掌圖》(1230 南宋) which reflected pronunciation at the time better)}\\
 
\end{frame}

\begin{frame}[fragile]{Early MC initials}

Some other notes:\\
\begin{itemize}
    \item in Canto., 清 voiceless = high tone, 濁 voiced = low tone
    \item Traditional \textbf{並} = Simplified \textbf{并}; but\\
    Mainland China writes MC initial [b] as \textbf{並}\\
    (Simp. \textbf{并} = Trad. \textbf{並} MC [b] or \textbf{併} [p/b] or \textbf{并} [p])
    \item 俟 MC [ʐ] only has two characters (俟漦) in 切韻, both merged into 崇 [ɖʐ] in 廣韻
    \item {\scriptsize (excluding 俟)} in Cantonese, the names of the retroflex initials 莊初崇生 have back vowels: zong1 co1 sung4 sang1; the names of the palatal initials 章昌常書船 have front vowels: zoeng1 coeng1 soeng4 syu1 syun4
\end{itemize}

\end{frame}
