漢語方言 \emph{hon3 jyu5 fong1 jin4 / hànyǔ fāngyán}: \\ \enquote{Sinitic languages} / \enquote{Chinese dialects} 

\bigskip

\underline{「方言」 partially overlaps with \enquote{dialect}} \\
方言：regional speech variety (language or dialect) \\
\quad Mair (1991): \enquote{topolect} as translation of 方言 \\
\quad \emph{mutual intelligibility not taken into account} \\
\quad \emph{usually used in opposition to a standard speech variety} \\ 
\textcolor{black!20}{dialect: speech varieties of a location\\
\quad language: collection of \enquote{mutually intelligible} dialects}

\vfill{\tiny \setstretch{1}Mair, Victor H. 1991. What Is a Chinese \enquote{Dialect/Topolect}? Reflections on Some Key Sino-English Linguistic Terms. \emph{Sino-Platonic Papers} 29. 1–31.\\}

\end{frame}
