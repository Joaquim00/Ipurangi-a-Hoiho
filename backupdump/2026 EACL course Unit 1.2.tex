\section{Cantonese 2}

\begin{frame}[fragile]{Cantonese}
\emph{One linguistic definition of Cantonese:} \\
\bigskip
De Sousa (2022)
\begin{itemize}
    \item Yuè dialect of Canton (\textsuperscript{one type of} \enquote{Cantonese})
    \item enclave Yuè varieties (\enquote{Cantonese enclaves}) resulted from the migration of people from the Canton area, or from another Cantonese enclave, primarily after the First Opium War (1840–1842)
\end{itemize}

\vfill{\tiny \setstretch{1} de Sousa, Hilário. 2022. The Expansion of Cantonese Over the Last Two Centuries. In Ye, Zhengdao (ed.), \emph{The Palgrave Handbook of Chinese Language Studies}, 409–440. Singapore: Springer Nature Singapore. \url{https://doi.org/10.1007/978-981-16-0924-4_35}\\}

\end{frame}

\begin{frame}[fragile]{Cantonese}
Qīng 清 Dynasty (1644 \textsuperscript{in Beijing} – 1912) \\ 
\quad before the First Opium War (1840–1842) \\
海禁 \emph{hoi2 gam3} / \emph{hǎijìn} \enquote*{sea ban}, e.g.:
\begin{itemize}
    \item restrictions on civilian ownership of watercrafts
    \item foreigners were only allowed to trade in \\Canton, Amoy (Xiàmén), Níngbō, Shànghǎi (1685-1757), \\and then only in Canton (1757-1842)
\end{itemize}
\bigskip
\emph{(¥¥¥¥ People of Canton were very rich ¥¥¥¥)}
\end{frame}

\begin{frame}[fragile]{Cantonese}

After the war (1842), China was forced to cancel the ban on civilian ownership of watercrafs.\\
\smallskip
Cantonese {\scriptsize (and other)} people migrated in large numbers by watercrafts \emph{within their river basin}, outward along the \emph{sea coast}, or \emph{overseas} (in multiple waves).\\

\end{frame} 

\begin{frame}[fragile]{Cantonese}

\emph{¥¥¥ People of Canton ¥¥¥}, through their business prowess, dominated many city and town centres in the Pearl River Region, {\scriptsize(and many Chinatowns overseas,)} forming many Cantonese enclaves.\\
\medskip
{\scriptsize Examples of Cantonese enclaves:\\
\quad Hong Kong, Macau, Zhànjiāng 湛江 (Fort-Bayard), Běihǎi 北海, Nánníng 南寧} \\
\bigskip
{\small Cantonese enclaves: observably different from the surrounding pre-existing dialects/languages; different language contact environments}

\end{frame}
