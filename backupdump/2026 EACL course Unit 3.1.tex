\section{Early Cantonese}

\begin{frame}[fragile]{Early Cantonese}
Earliest text with elements identifiable as Cantonese:\\
《花箋記》faa1 zin1 gei3\\ (late Ming: 17\textsuperscript{th} century, or perhaps 16\textsuperscript{th} century)

\end{frame}

\begin{frame}[fragile]{Early Cantonese}
Earliest Cantonese rime book:\\ 
《分韻撮要》fan1 wan5 cyut3 jiu3 (early Qing dynasty)\\
\smallskip
{\footnotesize Most-famous dated editions: 《江湖尺牘分韻撮要合集》(1782)\\《英華分韻撮要》 Tonic Dictionary of the Chinese Language in the Canton Dialect (1856)}\\
\medskip
Only recorded literary words / pronunciations\\
\medskip
\textit{\small (Many older traits of Standard Cantonese can still be found in other dialects / languages in the wider region)}

\end{frame}
