\section{Rimes}

\begin{frame}[fragile]{Codas}

\textbf{-p -m -t -n -k -ng -i -u}\\
/-p -m -t -n -k -ŋ -j -w/\\

\bigskip
\begin{itemize}
    \item /-p -t -k/: unreleased [p̚  t̚  k̚]
    \item highly stigmatised: errors with /-p -m/ 
    \item stigmatised: total or partial merger of /-k -ŋ/ into /-t -n/\\
    e.g. 香港 [hœŋ˥ kɔŋ˨˥] > [hœn˥ kɔn˨˥], 北角 [pɐk˥ kɔk˧] > [pɐt˥ kɔt˧]
    \item newer minor trends: \enquote{indeterminate} merger of /-n -ŋ/ as something like Ṽ, and /-t -k/ as something like ʔ\\
    e.g. 公安 [kʊŋ˥ ɔn˥] > [kʊ̃ ɔ̃], 失約 [sɐt˥ jœk˧] > [sɐʔ jœʔ]
\end{itemize}

\end{frame}

\begin{frame}[fragile]{Rimes}

粵拼 Jyut6 Ping3:
\begin{itemize}
    \item /y/ is rendered \textcolor{blue}{\textbf{yu}} (/j-/ is \textbf{j-}), e.g. 雨 \enquote*{rain} \textbf{jyu5} /jy˩˧/ [ɥy˩˧] 
    \item only contrastive \textcolor{blue}{long}–\textcolor{red}{short} vowel pairs are distinguished:
        \begin{itemize}
            \item \textcolor{blue}{\textbf{aa} /a/} vs. \textcolor{red}{\textbf{a} /ɐ/}
            \item \textcolor{blue}{\textbf{oe} /œ/} vs. \textcolor{red}{\textbf{eo} /ɵ/} {\scriptsize \emph{(very marginally contrastive)}}
        \end{itemize}\\
        (otherwise not distinguished, e.g. \textcolor{blue}{\textbf{it} [it]} vs. \textcolor{red}{\textbf{ik} [ek]})
    \item (unlike Dutch) \textcolor{blue}{long /a/} in open syllable is still spelt \textcolor{blue}{\textbf{aa}}
\end{itemize}
More generally:
\begin{itemize}
    \item vowel in open syllable has to be \textcolor{blue}{long}\\
    \quad (see below for the exceptional \textcolor{red}{short \textbf{a} [ɐ]})\\
    like e.g. most Germanic and Kra-Dai languages\\
    \quad \emph{\scriptsize short vowels in open syllables in Thai are later inventions}
\end{itemize}



\end{frame}
