\\\

\underline{「方言」 partially overlaps with \enquote{dialect}} \\
方言：regional speech variety (language or dialect) \\
\quad \emph{mutual intelligibility not taken into account} \\
\quad \emph{usually used in opposition to a standard speech variety} \\ 
\textcolor{black!20}{dialect: speech varieties of a location\\
\quad language: collection of \enquote{mutually intelligible} dialects}

\vfill{\tiny Mair, Victor H. 1991. What Is a Chinese \enquote{Dialect/Topolect}? Reflections on Some Key Sino-English Linguistic Terms. \emph{Sino-Platonic Papers} 29. 1–31.}

\section{Cantonese}

\begin{frame}[fragile]{Cantonese}

Cantonese < Canton \\
English \emph{\underline{C}anton} \textsuperscript{\emph{not canton}} < Portuguese \emph{Cantão} \\

\medskip
Português: \textsuperscript{\emph{since 16\textsuperscript{th} century}} \\
A província de Cantão 廣東 \\ 
A cidade de Cantão 廣州 \\
\smallskip
English: \\
\textcolor{gray!75}{The province of Canton 廣東
\textsuperscript{\emph{this use gradually faded away}}} \\
The \underline{city} of Canton 廣州

\end{frame}

\begin{frame}[fragile]{Cantonese}

\enquote{Cantonese} as an ethnic or linguistic term: \\
\enquote{Cantonese} \textsuperscript{\emph{usually}}≠ \enquote {related to the Province of 廣東}

\medskip
Sinitic dialect groups:
\begin{itemize}
    \item \large Yuè 粵
    \item Hakka 客家
    \item Mǐn 閩
    \item \small Unclassified Patois of N Guangdong 粵北土話
    \item \small Mandarin: Jūnhuà 軍話, SW Mandarin 西南官話
\end{itemize}
\small (+ 1\textasciitilde 1.5\%(?) of the population speak native non-Sinitic languages)

\end{frame}
